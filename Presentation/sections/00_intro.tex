\section{Task and Data}

\begin{frame}{The task: 2AFC categorization of vibration amplitude}
  \begin{itemize}
    \item stimulus $x_\text{stim} \in \{6, 12, 18, 24, 32, 38, 44, 50 \} \mu m$
    \begin{itemize}
      \item administered to one finger of one hand (left or right)
    \end{itemize}
    \item \textcolor{muted!60}{distractor $x_\text{dist} \in \{0, x_\text{d}\} \mu m$}
    \begin{itemize}
      \item \textcolor{muted!60}{$x_\text{d} \in \{6, 12, 18, 24, 32\} \mu m$ value fixed for each session}
      \item \textcolor{muted!60}{administered to the other (non-stimulus) hand}
    \end{itemize}
    \item \textcolor{muted!60}{if present, distractor administered simultaneously with the stimulus}
    \item monkey categorizes the stimulus as either
          ``high'' or ``low'' relative to a trained boundary of 28~$\mu m$.
    \begin{itemize}
      \item "low" response: correct for $x_\text{stim} < 28 \mu m$
      \item "high" response: correct for $x_\text{stim} > 28 \mu m$.
    \end{itemize}    
    \item correct responses are rewarded with water or juice
  \end{itemize}
\end{frame}

\begin{frame}{Psychometric function: response $r \sim \text{Bernoulli}(p = \psi)$}
  \vspace{-0.5em}
  \[
    \psi(x_\text{stim}) = \gamma_h + (1 - \gamma_h - \gamma_l)\,
                          \invlogit(\beta_0 + \beta_1 x_\text{stim})
  \]
  \centering
  \includegraphics[width=\textwidth,height=0.74\textheight,keepaspectratio]{example_psych_func.png}
\end{frame}

\begin{frame}{Parameters}
  \[
    \psi(x_\text{stim}) = \gamma_h + (1-\gamma_h-\gamma_l)\,
                          \invlogit(\beta_0 + \beta_1 x_\text{stim}),
    \qquad \vec{\theta} = (\beta_0, \beta_1, \gamma_h, \gamma_l)
  \]
  \begin{itemize}
    \item ${\color{blue}\gamma_h}$: lapse rate for ``high''
    \begin{itemize}
      \item lower bound on $\mathbf{P}(\text{guess} = \text{``high''})$, even to the weakest stimuli
      \item lower (left) asymptote of $\psi$
    \end{itemize}
    \item ${\color{purple}\gamma_l}$: lapse rate for ``low''
    \begin{itemize}
      \item lower bound on $\mathbf{P}(\text{guess} = \text{``low''})$, even to the strongest stimuli
      \item $1-\gamma_l$ = upper (right) asymptote of $\psi$
    \end{itemize}
    \item $\invlogit$: the logistic sigmoid, $\invlogit(z) = \dfrac{1}{1+\exp(-z)}$
    \item PSE: stimulus where responses of ``high'' and ``low'' are equally likely
    \begin{itemize}
      \item $x_\text{stim}$ such that $\psi(x_\text{stim}) = 0.5$
      \item represents bias
    \end{itemize}
    \item JND: how much the stimulus needs to change to be noticed half the time
    \begin{itemize}
      \item half the distance between $\psi(x_\text{stim}) = 0.25$ and $0.75$
      \item represents sensitivity
    \end{itemize}
  \end{itemize}
\end{frame}
