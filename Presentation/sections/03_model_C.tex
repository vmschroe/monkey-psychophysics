\begin{frame}{Model C: likelihood}
  For each hand, fit independently to its $n \approx 6{,}470$ unimanual and bimanual trials,
  with the distractor as a covariate $d_i$ (amplitude $/\,14.63$; $d_i = 0$ on unimanual trials):\\[0.3em]
  Likelihood: probability of obtaining responses $\vec{r}$ given parameters
  $\vec{\theta} = (\vec{\theta}_s)_{s=1}^{42}$,
  $\vec{\theta}_s = (\beta_{0,s}, \beta_{1,s}, \beta_{2,s}, \gamma_{h,s}, \gamma_{l,s})$\\
  $r_i \sim \mathbf{Bernoulli}[\psi(x_i, d_i;\vec{\theta}_{s[i]})]$, with
  $\psi(x, d;\vec{\theta}) = \gamma_h + (1-\gamma_h-\gamma_l)\,\invlogit(\beta_0 + \beta_1 x + \beta_2 d)$, so:
  \[
    \mathbf{P}[r_i = z] = \psi(x_i, d_i;\vec{\theta}_{s[i]})^{z}\,
                          \big(1-\psi(x_i, d_i;\vec{\theta}_{s[i]})\big)^{1-z},
    \qquad z \in \{0, 1\}
  \]
  \begin{align*}
    L(\vec{\theta};\vec{r}) &= \prod_{i=1}^{n} \mathbf{P}[r_i \mid \vec{\theta}_{s[i]}]\\
    &= \prod_{s=1}^{42}\ \prod_{i:\, s[i]=s} \psi(x_i, d_i;\vec{\theta}_s)^{r_i}\,
       \big(1-\psi(x_i, d_i;\vec{\theta}_s)\big)^{1-r_i}
  \end{align*}
\end{frame}

\begin{frame}{Model C: priors}
  Session parameters drawn from the hand's population (non-centered in the code):
  \begin{align*}
    \beta_{k,s} &\sim \mathcal{N}(\mu_{\beta_k},\ \sigma_{\beta_k}),\ k = 0, 1, 2, &
    \gamma_{h,s} &= 0.25\, w_{h,s}, \quad w_{h,s} \sim \text{LogitNormal}(\mu_{\gamma_h},\ \sigma_{\gamma_h}) \\
    & & \gamma_{l,s} &= 0.25\, w_{l,s}, \quad w_{l,s} \sim \text{LogitNormal}(\mu_{\gamma_l},\ \sigma_{\gamma_l})
  \end{align*}
  Hyperpriors, the same for both hands; all but those of $\beta_2$ as in Model B:
  \begin{align*}
    \mu_{\beta_0} &\sim \mathcal{N}(0,\ 2), & \sigma_{\beta_0} &\sim \text{HalfNormal}(2), \\
    \mu_{\beta_1} &\sim \text{LogNormal}(\log 4,\ 0.5), & \sigma_{\beta_1} &\sim \text{Exponential}(1), \\
    \mu_{\beta_2} &\sim \mathcal{N}(0,\ 1), & \sigma_{\beta_2} &\sim \text{HalfNormal}(1), \\
    \mu_{\gamma_h},\ \mu_{\gamma_l} &\sim \mathcal{N}(-3,\ 1.5), &
    \sigma_{\gamma_h},\ \sigma_{\gamma_l} &\sim \text{HalfNormal}(0.5)
  \end{align*}
  \vspace{-0.6em}
  {\small
  \begin{itemize}
    \item $\mu_{\beta_2}$ centered at 0: the distractor could push responses either way
    \item Uni- and bimanual trials share slope and lapse rates: $\beta_2 d$ only shifts the curve
  \end{itemize}}
\end{frame}

\subsection{Synthetic Data}

% numbers from Model_C/Convergence_C_Report.tex (made by Convergence_C.py)
\begin{frame}{Model C on synthetic data: the study}
  \begin{itemize}
    \item True hyperparameters derived from Model B's fit to the real data, so the synthetic
          data resemble the real data (e.g.\ $\mu_{\beta_2} = 0.93$ left, 1.13 right)
    \item Each simulated session gets one distractor amplitude, drawn from the real sessions'
    \item The same two sweeps as for Model B, each run twice with new data:
    \begin{itemize}
      \item sessions: 5, 10, 20, 40, 80 sessions of 308 trials
      \item trials: 42 sessions of 20, 40, 80, 160, 320 trials
    \end{itemize}
    \item 20 fits: 2 divergences in total (in the two smallest datasets),
          $\hat{R} \le 1.008$, bulk ESS $\ge 532$
  \end{itemize}
\end{frame}

\begin{frame}{Model C on synthetic data: what is recoverable}
  At the size of the real data (42 sessions of about 308 trials), if the data follow Model C:
  \begin{itemize}
    \item \textbf{Recoverable:} per-session PSEs in both conditions and JNDs; all five hand-level
          means, including the distractor effect $\mu_{\beta_2}$; the between-session sds of
          $\beta_0$, $\beta_1$ and $\beta_2$
    \item \textbf{Weakly determined:} the individual session coefficients, especially $\beta_2$
          in sessions with a small distractor (HDI 2.6 wide at 6~$\mu m$, 1.2 at 32~$\mu m$),
          which are pulled toward the hand mean
    \item \textbf{Not identified:} $\sigma_{\gamma_h}$ and $\sigma_{\gamma_l}$; their posteriors
          reflect the prior
    \item \textbf{Not tested:} what happens when the data break Model C's assumptions (shared
          lapse rates and slope, a shift proportional to amplitude)
  \end{itemize}
\end{frame}

\begin{frame}{Model C: conclusions}
  \small
  \begin{itemize}
    \item Reliable fit: 2 divergences in 20{,}000 draws, $\hat{R} = 1.000$
          ($\text{elpd}_\text{loo} = -3217.0$, SE 65.7)
    \item Effect of the distractor (bimanual $-$ unimanual), posterior mean [95\% HDI]:
    \begin{itemize}
      \item \textbf{PSE:} lowered, by $-4.51$ [$-5.15$, $-3.86$]~$\mu m$ (left) and
            $-5.74$ [$-6.39$, $-5.11$]~$\mu m$ (right) on average over sessions; more the stronger
            the distractor: a typical session shifts $-1.23$ and $-1.66$~$\mu m$ at 6~$\mu m$,
            $-6.57$ and $-8.85$~$\mu m$ at 32~$\mu m$
      \item \textbf{JND, \boldmath$\gamma_h$, \boldmath$\gamma_l$:} shared by both conditions, so
            the model cannot show an effect on them
    \end{itemize}
    \item The misfit sits where its assumptions meet the data:
    \begin{itemize}
      \item shared lapse rates: 6 of 32 pooled proportions outside their intervals, all at weak
            stimuli
      \item a shift proportional to amplitude: for the right hand, even a 6~$\mu m$ distractor
            shifts responses more than Model C allows
    \end{itemize}
    \item[$\Rightarrow$] Which model describes the data best? Model comparison
  \end{itemize}
\end{frame}
