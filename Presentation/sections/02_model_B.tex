\begin{frame}{Model B: likelihood}
  For each group (left/right $\times$ unimanual/bimanual), fit independently to its
  $n \approx 3{,}230$ trials, now with a parameter set for every session $s$:\\[0.3em]
  Likelihood: probability of obtaining responses $\vec{r}$ given parameters
  $\vec{\theta} = (\vec{\theta}_s)_{s=1}^{42}$,
  $\vec{\theta}_s = (\beta_{0,s}, \beta_{1,s}, \gamma_{h,s}, \gamma_{l,s})$\\
  $r_i \sim \mathbf{Bernoulli}[\psi(x_i;\vec{\theta}_{s[i]})]$ ($s[i]$: the session of trial $i$), so:
  \[
    \mathbf{P}[r_i = z] = \psi(x_i;\vec{\theta}_{s[i]})^{z}\,
                          \big(1-\psi(x_i;\vec{\theta}_{s[i]})\big)^{1-z},
    \qquad z \in \{0, 1\}
  \]
  \begin{align*}
    L(\vec{\theta};\vec{r}) &= \prod_{i=1}^{n} \mathbf{P}[r_i \mid \vec{\theta}_{s[i]}]\\
    &= \prod_{s=1}^{42}\ \prod_{i:\, s[i]=s} \psi(x_i;\vec{\theta}_s)^{r_i}\,
       \big(1-\psi(x_i;\vec{\theta}_s)\big)^{1-r_i}
  \end{align*}
\end{frame}

\begin{frame}{Model B: priors}
  Session parameters drawn from the group's population (non-centered in the code):
  \begin{align*}
    \beta_{0,s} &\sim \mathcal{N}(\mu_{\beta_0},\ \sigma_{\beta_0}), &
    \gamma_{h,s} &= 0.25\, w_{h,s}, \quad w_{h,s} \sim \text{LogitNormal}(\mu_{\gamma_h},\ \sigma_{\gamma_h}) \\
    \beta_{1,s} &\sim \mathcal{N}(\mu_{\beta_1},\ \sigma_{\beta_1}), &
    \gamma_{l,s} &= 0.25\, w_{l,s}, \quad w_{l,s} \sim \text{LogitNormal}(\mu_{\gamma_l},\ \sigma_{\gamma_l})
  \end{align*}
  Hyperpriors, the same in every group:
  \begin{align*}
    \mu_{\beta_0} &\sim \mathcal{N}(0,\ 2), & \sigma_{\beta_0} &\sim \text{HalfNormal}(2), \\
    \mu_{\beta_1} &\sim \text{LogNormal}(\log 4,\ 0.5), & \sigma_{\beta_1} &\sim \text{Exponential}(1), \\
    \mu_{\gamma_h},\ \mu_{\gamma_l} &\sim \mathcal{N}(-3,\ 1.5), &
    \sigma_{\gamma_h},\ \sigma_{\gamma_l} &\sim \text{HalfNormal}(0.5)
  \end{align*}
  \vspace{-0.6em}
  {\small
  \begin{itemize}
    \item Chosen from the design, checked by prior predictive simulation, not fit to the data
    \item $\mu_{\beta_0}$: typical PSE within 13--43~$\mu m$ with 90\% probability
    \item $\mu_{\beta_1} > 0$: median 4, 95\% in $[1.5,\ 10.6]$; $\sigma_{\beta_1}$ keeps 98\% of
          session slopes positive
    \item Typical lapse rate: median 0.012, 95\% in $[0.0007,\ 0.12]$
  \end{itemize}}
\end{frame}

\subsection{Synthetic Data}

\begin{frame}{Model B on synthetic data: the study}
  \begin{itemize}
    \item Fixed true hyperparameters for 4 synthetic groups, chosen to cover high and low values
          of each parameter (next slide); session parameters drawn from them
    \item Two sweeps, each run twice with new data:
    \begin{itemize}
      \item sessions: 5, 10, 20, 40, 80 sessions of 308 trials (tests the hyperparameters)
      \item trials: 42 sessions of 20, 40, 80, 160, 320 trials (tests the session parameters)
    \end{itemize}
    \item The real data (42 sessions of about 308 trials) are closest to the 40-session and
          320-trial fits
    \item All 20 fits sampled well: no divergences, $\hat{R} \le 1.010$, bulk ESS $\ge 579$
  \end{itemize}
\end{frame}

\begin{frame}{Model B on synthetic data: what is recoverable}
  At the size of the real data (42 sessions of about 308 trials):
  \begin{itemize}
    \item \textbf{Recoverable:} per-session PSE and JND; the bias mean $\mu_{\beta_0}$ ($\pm 1$);
          the slope mean $\mu_{\beta_1}$ when the slope is shallow (3); the bias spread
          $\sigma_{\beta_0}$ (0.25--1.0); $\mu_{\gamma_h}$ in most groups
    \item \textbf{Weakly constrained:} $\mu_{\beta_1}$ when the slope is steep (6): the curve rises
          within one stimulus spacing, so only 2 levels inform it (HDIs 1.8--2.6 wide)
    \item \textbf{Biased in some datasets:} a lapse rate whose side of a shallow curve never
          reaches its asymptote: $\gamma_l$ too high when the PSE is high, $\gamma_h$ too low when
          it is low; the higher $\mu_{\gamma_l}$ ($-2.8$) is also overestimated at 80 sessions
    \item \textbf{Not identified:} the sds with true values near 0 ($\sigma_{\beta_1}$,
          $\sigma_{\gamma_h}$, $\sigma_{\gamma_l}$); their posteriors reflect the prior
    \item Session-level HDIs mostly conservative; PSE coverage slightly low (0.90--0.94)
  \end{itemize}
\end{frame}

\begin{frame}{Model B: conclusions}
  \small
  \begin{itemize}
    \item Reliable fit; no systematic misfit, pooled or within sessions
          ($\text{elpd}_\text{loo} = -3180.0$, SE 64.3)
    \item Effect of the distractor (bimanual $-$ unimanual), average over the 42 sessions,
          posterior mean [95\% HDI]:
    \begin{itemize}
      \item \textbf{PSE:} lowered, by $-4.29$ [$-5.05$, $-3.52$]~$\mu m$ (left) and
            $-6.06$ [$-6.80$, $-5.34$]~$\mu m$ (right); negative in 40/42 and 42/42 sessions
      \item \textbf{JND:} raised for the left hand, $+1.03$ [0.46, 1.59]~$\mu m$, driven by a
            few sessions; unclear for the right, $+0.49$ [$-0.07$, 1.03]~$\mu m$
      \item \textbf{\boldmath$\gamma_h$:} raised, by $+0.025$ [0.009, 0.042] (left) and
            $+0.035$ [0.016, 0.054] (right)
      \item \textbf{\boldmath$\gamma_l$:} no effect, $0.000$ [$-0.014$, 0.013] (left) and
            $+0.003$ [$-0.009$, 0.015] (right)
    \end{itemize}
    \item The PSE shift grows with distractor amplitude, by about $-2$~$\mu m$ per 10~$\mu m$
          (exploratory; amplitude is confounded with session order)
    \item[$\Rightarrow$] Put the distractor amplitude into the model: Model C
  \end{itemize}
\end{frame}
