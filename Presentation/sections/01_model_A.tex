\begin{frame}{Model A: likelihood}
  For each group (left/right $\times$ unimanual/bimanual), fit independently to its
  $n \approx 3{,}230$ trials from all 42 sessions pooled:\\[0.3em]
  Likelihood: probability of obtaining responses $\vec{r}$ given parameters
  $\vec{\theta} = (\beta_0, \beta_1, \gamma_h, \gamma_l)$\\
  $r_i \sim \mathbf{Bernoulli}[\psi(x_i;\vec{\theta})]$ ($r_i = 1$ for ``high'', $x_i$ standardized), so:
  \[
    \mathbf{P}[r_i = z] = \psi(x_i;\vec{\theta})^{z}\,\big(1-\psi(x_i;\vec{\theta})\big)^{1-z},
    \qquad z \in \{0, 1\}
  \]
  \begin{align*}
    L(\vec{\theta};\vec{r}) &= \prod_{i=1}^{n} \mathbf{P}[r_i \mid \vec{\theta}]\\
    &= \prod_{i=1}^{n} \psi(x_i;\vec{\theta})^{r_i}\,\big(1-\psi(x_i;\vec{\theta})\big)^{1-r_i}
  \end{align*}
\end{frame}

\begin{frame}{Model A: priors}
  The same priors in every group:
  \begin{align*}
    \beta_0 &\sim \mathcal{N}(0,\ 6) \\
    \beta_1 &\sim \mathcal{N}(4,\ 2) \\
    \gamma_h &= 0.25\, w_h, \qquad w_h \sim \text{Beta}(1.22,\ 4.89) \\
    \gamma_l &= 0.25\, w_l, \qquad w_l \sim \text{Beta}(1.22,\ 4.89)
  \end{align*}
  \begin{itemize}
    \item $\beta_1$ must be positive for the psychometric function to have a positive slope
    \item $\gamma$'s have support between 0 and 0.25, with prior mean of 0.05 and 95\% range
          $[0.003,\ 0.141]$
  \end{itemize}
\end{frame}

\subsection{Synthetic Data}

% numbers from Model_A/Convergence_A_Report.tex (made by Convergence_A.py)
\begin{frame}{Model A on synthetic data: the study}
  \begin{itemize}
    \item One fixed set of true parameters $(\beta_0, \beta_1, \gamma_h, \gamma_l)$ per group
    \item Trials drawn from the real design (the real mix of stimulus levels and groups),
          responses simulated from the true curves
    \item 7 sizes: 250 to 16{,}000 trials; the real data (12{,}937) lie between the two largest
    \item 20 replicates of the whole sweep: 140 fits, 80 per size and parameter
    \item All fits sampled well: $\hat{R} \le 1.006$, bulk ESS $\ge 1{,}554$
  \end{itemize}
\end{frame}

\begin{frame}{Model A on synthetic data: what is recoverable}
  At the size of the real data (about 3{,}230 trials per group):
  \begin{itemize}
    \item \textbf{Recoverable:} PSE, JND, $\beta_0$, $\gamma_h$: unbiased, calibrated HDIs
    \begin{itemize}
      \item posterior sd about 0.3~$\mu m$ for the PSE, 0.2~$\mu m$ for the JND
    \end{itemize}
    \item \textbf{Recoverable, HDIs a little wide:} $\beta_1$ (coverage 0.98--0.99)
    \item \textbf{Not reliably recoverable:} small $\gamma_l$
    \begin{itemize}
      \item biased upward by 0.4--0.5 sd; by about 1 sd when the curve never reaches its
            ceiling (left unimanual), where $\gamma_l$ trades off with the slope
      \item HDIs still cover the truth, but the posterior mean should not be read literally
    \end{itemize}
    \item \textbf{Caveats:} one set of true values; data simulated with one curve per group,
          so this says nothing about variation between sessions
  \end{itemize}
\end{frame}

\begin{frame}{Model A: conclusions}
  \begin{itemize}
    \item Reliable fit; the pooled data are reproduced at every amplitude
          ($\text{elpd}_\text{loo} = -3344.0$, SE 64.5)
    \item Effect of the distractor (bimanual $-$ unimanual), posterior mean [95\% HDI]:
    \begin{itemize}
      \item \textbf{PSE:} lowered, by $-4.19$ [$-4.96$, $-3.44$]~$\mu m$ (left) and
            $-6.07$ [$-6.85$, $-5.30$]~$\mu m$ (right)
      \item \textbf{JND:} raised for the left hand, $+0.83$ [0.27, 1.37]~$\mu m$; unclear for
            the right, $+0.27$ [$-0.29$, 0.80]~$\mu m$
      \item \textbf{\boldmath$\gamma_h$:} raised, by $+0.038$ [0.023, 0.055] (left) and
            $+0.042$ [0.021, 0.061] (right)
      \item \textbf{\boldmath$\gamma_l$:} no clear effect, $+0.007$ [$-0.008$, 0.022] (left) and
            $+0.006$ [$-0.006$, 0.017] (right)
    \end{itemize}
    \item[$\Rightarrow$] A curve for every session: Model B
  \end{itemize}
\end{frame}
